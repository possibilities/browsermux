cmake_minimum_required(VERSION 3.21)
project(browsermuxNative LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "Native host requires the official macOS SDK. Only Rust tests run on Linux.")
endif()
set(PROJECT_ARCH "arm64")
set(USE_SANDBOX ON CACHE BOOL "Sandbox is mandatory" FORCE)
set(CEF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/cef" CACHE PATH "Pinned CEF distribution")
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
set(api_version "15400")
find_package(CEF REQUIRED)
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)
add_library(spb_engine STATIC CEF/BrowserEngine.mm)
target_include_directories(spb_engine PRIVATE "${CEF_ROOT}")
target_compile_definitions(spb_engine PRIVATE CEF_USE_SANDBOX CEF_API_VERSION=15400)
target_compile_features(spb_engine PRIVATE cxx_std_20)
target_compile_options(spb_engine PRIVATE -fobjc-arc -fno-rtti -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(spb_engine PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
add_executable(spb_helper CEF/HelperMain.cc)
target_include_directories(spb_helper PRIVATE "${CEF_ROOT}")
target_compile_definitions(spb_helper PRIVATE CEF_USE_SANDBOX CEF_API_VERSION=15400)
target_compile_features(spb_helper PRIVATE cxx_std_20)
target_link_libraries(spb_helper PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
